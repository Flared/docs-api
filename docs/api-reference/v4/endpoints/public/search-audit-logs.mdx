<PermissionReadOrganizationAdmin />

Returns a list of audit logs for the current organization, which can be used to stream all audit logs to an external platform or to search for specific ones.

## Body Parameters

<ParamField body="from" type="string">
  The `next` value from the last response.
</ParamField>

<ParamField body="order" type="string" default="desc">
  Available options: `asc`, `desc`
</ParamField>

<ParamField body="size" type="integer" default={50}>
</ParamField>

<ParamField body="filters" type="object">
  <Expandable title="child attributes">
    <ParamField body="affected_tenant_ids" type="integer[]">
      Tenants affected by the audit log.
    </ParamField>

    <ParamField body="created_at" type="object">
      <Expandable title="child attributes">
        <ParamField
          body="gte"
          type="string"
          placeholder="Example: 2024-01-01T00:00:00+00:00"
        >
          Matches values greater than or equal to the specified timestamp.

          Format: ISO-8601
        </ParamField>

        <ParamField
          body="lte"
          type="string"
          placeholder="Example: 2024-01-01T00:00:00+00:00"
        >
          Matches values lesser than or equal to the specified timestamp.

          Format: ISO-8601
        </ParamField>
      </Expandable>
    </ParamField>

    <ParamField body="query" type="string">
      Free text search on the audit log content.
    </ParamField>
  </Expandable>
</ParamField>
